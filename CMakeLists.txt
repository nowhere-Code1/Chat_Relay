cmake_minimum_required(VERSION 3.22)
project(ChatRelay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(RELAY_HOST "tiktok-bot-3ayj.onrender.com" CACHE STRING "Host name of the bot (no https://)")
add_library(chat_relay SHARED main.cpp)
set_target_properties(chat_relay PROPERTIES OUTPUT_NAME "chat_relay" PREFIX "lib" SUFFIX ".so")
target_compile_definitions(chat_relay PRIVATE RELAY_HOST="${RELAY_HOST}")
target_compile_options(chat_relay PRIVATE -fvisibility=hidden -Wall -Wextra)
target_link_libraries(chat_relay PRIVATE log)
target_link_options(chat_relay PRIVATE -Wl,-z,max-page-size=16384 -Wl,--build-id=sha1)
